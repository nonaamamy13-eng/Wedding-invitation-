          <div class="form-group">
            <label for="guest-note">رسالة تهنئة للعروسين (اختياري)</label>
            <input type="text" id="guest-note" class="form-control" placeholder="اكتب كلمة لطيفة للعروسين..." />
          </div>

          <button type="submit" class="submit-btn">إرسال تأكيد الحضور عبر واتساب</button>
        </form>

        <div id="rsvp-success">
          <h3>شكراً لك! 🤍</h3>
          <p id="success-message">تم تسجيل ردك بنجاح، نتطلع لرؤيتك ومشاركتنا الفرحة.</p>
        </div>

  <script>
    // 1. إعدادات عود الثقاب التفاعلي والشمعدانات
    const match = document.getElementById('draggable-match');
    const matchTip = document.getElementById('match-flame-tip');
    const dock = document.getElementById('match-dock');
    const introStage = document.getElementById('intro-stage');
    const invitationContent = document.getElementById('invitation-content');
    const hintText = document.getElementById('hint-text');

    const candles = [
      { unit: document.getElementById('candle-1'), wick: document.getElementById('wick-1'), lit: false },
      { unit: document.getElementById('candle-2'), wick: document.getElementById('wick-2'), lit: false }
    ];

    let isDragging = false;
    let litCount = 0;

    // وضع عود الثقاب في نقطة البداية داخل الحلقة
    function resetMatchPosition() {
      const dockRect = dock.getBoundingClientRect();
      const startX = dockRect.left + (dockRect.width / 2) - 25;
      const startY = dockRect.top + (dockRect.height / 2) - 26;
      match.style.left = `${startX}px`;
      match.style.top = `${startY}px`;
    }

    window.addEventListener('load', resetMatchPosition);
    window.addEventListener('resize', () => {
      if (!isDragging) resetMatchPosition();
    });

    // حساب المسافة بين لهب عود الثقاب وفتيل كل شمعة
    function checkCandleCollision() {
      const tipRect = matchTip.getBoundingClientRect();
      const tipX = tipRect.left + tipRect.width / 2;
      const tipY = tipRect.top + tipRect.height / 2;

      candles.forEach((candle) => {
        if (candle.lit) return;

        const wickRect = candle.wick.getBoundingClientRect();
        const wickX = wickRect.left + wickRect.width / 2;
        const wickY = wickRect.top + wickRect.height / 2;

        const distance = Math.hypot(tipX - wickX, tipY - wickY);

        // إذا اقترب اللهب لمسافة أقل من 42 بكسل تشتعل الشمعة
        if (distance < 42) {
          candle.lit = true;
          candle.unit.classList.add('is-lit');
          litCount++;

          // اهتزاز خفيف للهاتف عند إشعال الشمعة (مدعوم في أندرويد)
          if (navigator.vibrate) navigator.vibrate(45);

          if (litCount === 1) {
            hintText.textContent = 'رائع! أشعلي الشمعة الثانية الآن ✨';
          } else if (litCount === 2) {
            completeLighting();
          }
        }
      });
    }

    function completeLighting() {
      hintText.textContent = '✨ أضاءت ليلتنا بوجودكم... جاري فتح الدعوة ✨';
      introStage.classList.add('lit-glow');
      if (navigator.vibrate) navigator.vibrate([60, 50, 100]);

      setTimeout(() => {
        introStage.classList.add('hidden-stage');
        invitationContent.classList.add('visible');
        match.style.display = 'none';
      }, 1350);
    }

    // أحداث اللمس والماوس للسحب
    function onStart(e) {
      isDragging = true;
    }

    function onMove(e) {
      if (!isDragging) return;
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;

      // جعل طرف عود الثقاب فوق الإصبع قليلاً ليكون واضحاً أثناء السحب
      match.style.left = `${clientX - 18}px`;
      match.style.top = `${clientY - 38}px`;

      checkCandleCollision();
    }

    function onEnd() {
      if (!isDragging) return;
      isDragging = false;
      if (litCount < 2) {
        resetMatchPosition();
      }
    }

    match.addEventListener('mousedown', onStart);
    window.addEventListener('mousemove', onMove);
    window.addEventListener('mouseup', onEnd);

    match.addEventListener('touchstart', onStart, { passive: true });
    window.addEventListener('touchmove', onMove, { passive: true });
    window.addEventListener('touchend', onEnd);

    // زر التخطي
    document.getElementById('skip-intro').addEventListener('click', () => {
      introStage.classList.add('hidden-stage');
      invitationContent.classList.add('visible');
      match.style.display = 'none';
    });

    // 2. العداد التنازلي لموعد الزفاف (عدّلي التاريخ هنا)
    const weddingDate = new Date('2026-05-15T20:00:00').getTime();

    setInterval(() => {
      const now = new Date().getTime();
      const diff = weddingDate - now;

      if (diff <= 0) return;

      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
      const seconds = Math.floor((diff % (1000 * 60)) / 1000);

      document.getElementById('days').textContent = String(days).padStart(2, '0');
      document.getElementById('hours').textContent = String(hours).padStart(2, '0');
      document.getElementById('minutes').textContent = String(minutes).padStart(2, '0');
      document.getElementById('seconds').textContent = String(seconds).padStart(2, '0');
    }, 1000);

    // 3. إرسال تأكيد الحضور عبر واتساب
    document.getElementById('rsvp-form').addEventListener('submit', function(e) {
      e.preventDefault();

      const name = document.getElementById('guest-name').value.trim();
      const attendance = document.querySelector('input[name="attendance"]:checked').value;
      const count = document.getElementById('guests-count').value;
      const note = document.getElementById('guest-note').value.trim();

      // ضعي رقم واتساب منظم الحفل هنا بصيغة دولية بدون + أو أصفار في البداية
      const whatsappNumber = "218910000000";

      let message = `مرحباً، تأكيد حضور حفل الزفاف 💍\n`;
      message += `• الاسم: ${name}\n`;
      message += `• الحالة: ${attendance}\n`;
      if (attendance.includes('سأحضر')) {
        message += `• عدد الحضور: ${count}\n`;
      }
      if (note) {
        message += `• تهنئة: ${note}`;
      }

      const whatsappUrl = `https://wa.me/${whatsappNumber}?text=${encodeURIComponent(message)}`;

      this.style.display = 'none';
      document.getElementById('rsvp-success').style.display = 'block';
      document.getElementById('success-message').textContent = `شكراً لك يا ${name}، تم تجهيز رسالة التأكيد لإرسالها.`;

      window.open(whatsappUrl, '_blank');
    });
  </script>
